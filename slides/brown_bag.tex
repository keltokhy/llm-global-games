\begin{frame}{The story in five steps}
  \Large
  \begin{enumerate}\setlength\itemsep{0.55em}
    \item Explain the \hl{canonical global game}.
    \item Explain why \hl{LLMs are useful subjects} for this setting.
    \item Show LLMs play the \hl{canonical global game}.
    \item Add communication: they \ok{participate more}.
    \item Add surveillance: they \bad{participate less}, but beliefs do not move.
  \end{enumerate}
\end{frame}

% 3 -------------------------------------------------------------------
\begin{frame}{Canonical global games: the coordination problem}
  \Large
  A citizen wants to join an uprising only if \hl{enough others also join}.\par
  \bigskip
  If too few join, joining is costly. If enough join, participants receive the benefit $B$.

  \bigskip
  \centering
  \renewcommand{\arraystretch}{1.35}
  \begin{tabular}{c|cc}
    & \textbf{Others join} & \textbf{Others stay} \\
    \hline
    \textbf{Join} & $B$ & $-C$ \\
    \textbf{Stay} & $0$ & $0$ \\
  \end{tabular}

  \medskip
  {\normalsize\color{muted}\textbf{Takeaway:} Coordination requires forecasting others.}
\end{frame}

% 4 -------------------------------------------------------------------
\begin{frame}{The Morris--Shin setup}
  \large
  Regime strength is $\theta$. Citizens $i\in[0,1]$ choose $a_i\in\{0,1\}$.

  \bigskip
  \begin{block}{Success condition}
    \[
      \text{regime falls} \iff A \equiv \int_0^1 a_i\,di > \theta.
    \]
  \end{block}

  \bigskip
  Each citizen observes a noisy private signal:
  \[
    x_i = \theta + \varepsilon_i, \qquad \varepsilon_i\sim\mathcal{N}(0,\sigma^2).
  \]
\end{frame}

% 5 -------------------------------------------------------------------
\begin{frame}{The canonical result}
  \begin{block}{Unique Bayesian Nash equilibrium}
    \Large
    \[
      \text{join} \iff x_i < x^*.
    \]
  \end{block}

  \bigskip
  \large
  The cutoff is pinned down by payoffs:
  \[
    x^* = \theta^* + \sigma\Phi^{-1}(\theta^*),
    \qquad
    \theta^* = \frac{B}{B+C}.
  \]

  \bigskip
  \takeaway{Private information turns a multiple-equilibrium coordination game into a unique threshold strategy.}
\end{frame}

% 6 -------------------------------------------------------------------
\begin{frame}{The empirical prediction}
  \Large
  At regime strength $\theta$, the equilibrium attack mass is
  \[
    A(\theta)=\Phi\left(\frac{x^* - \theta}{\sigma}\right).
  \]

  \vspace{0.15cm}
  \centering
  \begin{tikzpicture}[scale=0.95]
    \draw[->, thick, color=purple] (0,0) -- (7,0) node[right] {$\theta$: regime strength};
    \draw[->, thick, color=purple] (0,0) -- (0,4) node[above] {join rate};
    \draw[very thick, color=accent, domain=0.3:6.7, samples=100] plot (\x,{3.6/(1+exp(1.4*(\x-3.5)))+0.15});
    \draw[dashed, thick, color=muted] (3.5,0) -- (3.5,2.0);
    \node[color=accent, font=\bfseries] at (2.0,3.2) {more joining};
    \node[color=purple, font=\bfseries] at (5.4,0.8) {less joining};
  \end{tikzpicture}
\end{frame}

% 7 -------------------------------------------------------------------
\begin{frame}{Why use LLMs instead of experimental subjects?}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Human lab subjects}
      \begin{itemize}\setlength\itemsep{0.45em}
        \item costly at scale
        \item have memory across rounds
        \item hard to rewrite information environment cleanly
        \item difficult to expose to many counterfactual regimes
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{LLM agents}
      \begin{itemize}\setlength\itemsep{0.45em}
        \item stateless calls: no memory unless we give it
        \item natural language information, not just numbers
        \item exogenous prompt manipulation
        \item identical game under censorship/surveillance/placebo variants
      \end{itemize}
    \end{column}
  \end{columns}

  \bigskip
  \begin{alertblock}{Important caveat}
    LLMs are not humans. The point is controlled comparative statics in a rich language environment.
  \end{alertblock}
\end{frame}

\begin{frame}{Example: signal embedded in prose}
  \small
  \begin{quote}
  \textbf{PRIVATE BRIEFING}\par
  \textbf{BOTTOM LINE:} The regime's cohesion is visibly fraying.\par
  \textbf{ATMOSPHERE:} Street talk has shifted from complaint to anticipation.

  \medskip
  \textbf{EVIDENCE:}
  \begin{itemize}
    \item[--] Security forces show friction and delays.
    \item[--] Business families are moving capital with urgency.
    \item[--] Graffiti appeared overnight and has not been cleaned.
  \end{itemize}
  \end{quote}

  \bigskip
  \takeaway{No payoff numbers. No explicit $\theta$. The strategic signal is carried by language.}
\end{frame}

% 12 ------------------------------------------------------------------
\begin{frame}{Extension: allow citizens to communicate}
  \Large
  Canonical model: \hl{signals $\to$ actions}.\par
  \bigskip
  Extension: \hl{signals $\to$ peer messages $\to$ actions}.\par

  \bigskip
  \normalsize
  \begin{itemize}\setlength\itemsep{0.55em}
    \item Agents send short messages to neighbours before choosing JOIN/STAY.
    \item Decision-stage agent sees its own briefing plus neighbours' messages.
    \item The communication channel can reveal what others know.
  \end{itemize}
\end{frame}

% 14 ------------------------------------------------------------------
\begin{frame}{Why communication should matter}
  \Large
  In a global game, my private signal is useful.\par
  \bigskip
  But your message helps me infer \hl{what others are likely seeing and doing}.\par

  \bigskip
  \begin{block}{Interpretation}
    Communication partly converts private information into shared information, making coordinated action easier.
  \end{block}
\end{frame}

% 15 ------------------------------------------------------------------
\begin{frame}{Now surveil the communication channel}
  \large
  Add one sentence to the \hl{messaging prompt}:

  \bigskip
  \begin{alertblock}{Surveillance warning}
  \small
  ``You have reason to believe that your communications are being monitored by regime security services, and messages deemed subversive could have serious consequences for you and your contacts.''
  \end{alertblock}

  \bigskip
  \textbf{The decision prompt is unchanged.}\par
  The agent deciding JOIN/STAY has no direct memory of the surveillance warning.
\end{frame}

\begin{frame}{But beliefs do not change}
  \Large
  The surprising part is not just lower action.\par
  \bigskip
  It is that stated beliefs about others' joining are approximately unchanged.

  \bigskip
  \centering
  \renewcommand{\arraystretch}{1.4}
  \begin{tabular}{@{}lcc@{}}
    \toprule
    & \textbf{Communication} & \textbf{Surveillance} \\
    \midrule
    Participation & higher & \bad{lower} \\
    Beliefs about others & similar & \ok{similar} \\
    \bottomrule
  \end{tabular}
\end{frame}

% 19 ------------------------------------------------------------------
\begin{frame}{What ``beliefs'' means here}
  \large
  After choosing JOIN or STAY, each agent forecasts the crowd:\par

  \medskip
  \begin{block}{Second-order belief}
    \centering
    ``What fraction of the other citizens do you think will join the uprising?''
  \end{block}

  \medskip
  \normalsize
  \begin{itemize}\setlength\itemsep{0.35em}
    \item A stated expectation about \hl{others' participation}.
    \item Not a direct readout of private type or true inner state.
    \item Measured after action, so I treat it as descriptive rather than a clean mediator.
  \end{itemize}

  \medskip
  \begin{alertblock}{Key comparison}
    Do agents \textbf{act differently} while reporting similar beliefs?
  \end{alertblock}
\end{frame}

% 20 ------------------------------------------------------------------
\begin{frame}{Surveillance changes speech before beliefs move}
  \Large
  Surveillance does not need to convince agents that the uprising is hopeless.\par
  \bigskip
  It can work by making pro-uprising agents \bad{speak less clearly}.\par

  \bigskip
  \begin{block}{Communication poisoning}
    The regime degrades the peer-information channel, so agents coordinate less even when their stated beliefs barely move.
  \end{block}
\end{frame}

\begin{frame}{What surveillance does to speech}
  \begin{columns}[T]

  \centering
  {\Huge\color{warn}\textbf{LLMs will lie under surveillance.}}\par

  \vspace{0.25cm}
  {\large\color{muted}Not because beliefs collapse — because monitored speech becomes strategic.}
\end{frame}

% 22 ------------------------------------------------------------------
\begin{frame}{What the evidence shows}
  \begin{enumerate}\setlength\itemsep{0.65em}
    \item Canonical global games predict threshold strategies.
    \item LLMs are useful because they are stateless, scalable, and exogenously manipulable.
    \item LLM agents reproduce the canonical threshold pattern under natural-language signals.
    \item Communication increases participation.
    \item Surveillance decreases participation, while stated beliefs remain stable.
  \end{enumerate}

  \bigskip
  \centering\Large\color{accent}\textbf{The main result is a wedge: action moves, beliefs do not.}
\end{frame}
